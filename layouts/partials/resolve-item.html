{{- $base := .base | default "" -}} {{- $file := .file -}} {{- $root := $base | replaceRE "//+" "/" -}} {{- $folder := printf "static/%s" $root -}} {{- if not (fileExists $folder) -}} {{- printf "debug|missing_folder|%s\n" $folder -}} {{- return -}} {{- end -}} {{- $files := readDir $folder -}} {{- range $files -}} {{- $name := .Name -}} {{- if strings.HasPrefix $name $file -}} {{- $ext := lower (path.Ext $name) -}} {{- $full := printf "%s/%s" $root $name -}} {{- if or (eq $ext ".jpg") (eq $ext ".png") -}} {{ printf "image|%s\n" $full }} {{- else if eq $ext ".pdf" -}} {{ printf "pdf|%s\n" $full }} {{- else if eq $ext ".mp3" -}} {{ printf "audio|%s\n" $full }} {{- else if eq $ext ".md" -}} {{ printf "text|%s\n" $full }} {{- else -}} {{ printf "file|%s\n" $full }} {{- end -}} {{- end -}} {{- end -}}
